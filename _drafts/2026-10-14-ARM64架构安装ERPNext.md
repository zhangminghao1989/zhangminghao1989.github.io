本文记录在ARM64架构Ubuntu 24.04系统上搭建ERPNext的过程
<!--more-->

克隆这个仓库
git clone https://github.com/frappe/frappe_docker
cd frappe_docker

# 部署docker

编辑frappe.env文件
cp example.env frappe.env
修改内容
ERPNEXT_VERSION=v16.50.0
DB_PASSWORD=admin
DB_HOST=db
DB_PORT=3306
REDIS_CACHE=redis-cache:6379
REDIS_QUEUE=redis-queue:6379
REDIS_SOCKETIO=redis-socketio:6379
LETSENCRYPT_EMAIL=None
SITE_ADMIN_PASS=123
SITES_RULE=Host(`erp.example.com`)
PULL_POLICY=missing
BACKUP_CRONSTRING="@every 6h"
HTTP_PUBLISH_PORT=8080


生成compose文件
docker compose --env-file frappe.env \
    -f compose.yaml \
    -f overrides/compose.mariadb.yaml \
    -f overrides/compose.redis.yaml \
    -f overrides/compose.noproxy.yaml \
    -f overrides/compose.backup-cron.yaml \
    config > frappe-compose.yaml

将frappe-compose.yaml中的platform: linux/amd64改为platform: linux/arm64

将两个文件复制到用户目录
拉取镜像
sudo docker compose -p frappe -f /home/ubuntu/frappe-compose.yml pull
生成容器
sudo docker compose -p frappe -f /home/ubuntu/frappe-compose.yml up --force-recreate --remove-orphans -d

生成网站
docker compose -p frappe exec backend bench new-site erp.example.com --mariadb-user-host-login-scope='172.%.%.%'
docker compose -p frappe exec backend bench --site erp.example.com install-app erpnext

# 其他用到的命令
安装docker
curl -fsSL https://get.docker.com -o install-docker.sh
cat install-docker.sh
sh install-docker.sh --dry-run
sudo sh install-docker.sh


打开防火墙端口
sudo apt install ufw
sudo ufw enable
sudo ufw allow OpenSSH
sudo ufw allow 'Nginx Full'

更新frappe_docker
将本地代码放到暂存区
git stash
拉取远程代码
git pull
恢复暂存区的代码
git stash pop

删除已经创建的docker
sudo docker compose -p frappe -f /home/ubuntu/frappe-compose.yml down -v
sudo docker volume prune -a
